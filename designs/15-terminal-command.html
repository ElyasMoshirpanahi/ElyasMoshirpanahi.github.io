# module insurers/risk-api

Underwrite with signals, not surveys

A programmable stream of anonymized, consent-based population health data. One endpoint, actuarial-grade output.

// FitInsured Risk API — Node
import { FitInsured } from '@fitinsured/sdk';

const fi = FitInsured({ apiKey: process.env.FI_KEY });

const risk = await fi.cohorts.risk({
  segment: 'active-40-49',
  anonymity: 'k50',
  window: '90d'
});
